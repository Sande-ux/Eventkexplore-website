        // --- 3. COUNTDOWN TIMER LOGIC ---
        function startFlagshipCountdown() {
            const targetDate = new Date("2026-11-20T09:00:00").getTime();

            function update() {
                const now = new Date().getTime();
                const diff = targetDate - now;

                if (diff <= 0) {
                    document.getElementById('hero-days').innerText = "00";
                    document.getElementById('hero-hours').innerText = "00";
                    document.getElementById('hero-mins').innerText = "00";
                    document.getElementById('hero-secs').innerText = "00";
                    return;
                }

                const days = Math.floor(diff / (1000 * 60 * 60 * 24));
                const hours = Math.floor((diff % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
                const mins = Math.floor((diff % (1000 * 60 * 60)) / (1000 * 60));
                const secs = Math.floor((diff % (1000 * 60)) / 1000);

                document.getElementById('hero-days').innerText = days.toString().padStart(2, '0');
                document.getElementById('hero-hours').innerText = hours.toString().padStart(2, '0');
                document.getElementById('hero-mins').innerText = mins.toString().padStart(2, '0');
                document.getElementById('hero-secs').innerText = secs.toString().padStart(2, '0');
            }
            update();
            setInterval(update, 1000);
        }

            document.getElementById('modal-category').innerText = evt.category;
            document.getElementById('modal-date').innerText = evt.displayDate;
            document.getElementById('modal-time').innerText = evt.time;
            document.getElementById('modal-venue').innerText = evt.venue;

            // Agenda List

            agendaContainer.innerHTML = '';
            evt.agenda.forEach(item => {

            // Clear errors
            document.getElementById('err-name').classList.add('hidden');

            document.getElementById('err-phone').classList.add('hidden');

            document.getElementById('event-modal').classList.remove('hidden');
        }

            document.getElementById('event-modal').classList.add('hidden');
        }

        function handleRegistrationSubmit(e) {
            e.preventDefault();
            
            const eventId = parseInt(document.getElementById('form-event-id').value);
            const evt = EVENTS_DATA.find(item => item.id === eventId);
            const name = document.getElementById('reg-name').value.trim();
            const email = document.getElementById('reg-email').value.trim();
            const phone = document.getElementById('reg-phone').value.trim();
            const ticketTier = document.getElementById('reg-ticket-type').value;

            let valid = true;

            if (!name) {
                document.getElementById('err-name').classList.remove('hidden');
                valid = false;
            } else {
                document.getElementById('err-name').classList.add('hidden');
            }

            const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            if (!emailRegex.test(email)) {
                document.getElementById('err-email').classList.remove('hidden');
                valid = false;
            } else {
                document.getElementById('err-email').classList.add('hidden');
            }

            const phoneRegex = /^[0-9]{10}$/;
            if (!phoneRegex.test(phone)) {
                document.getElementById('err-phone').classList.remove('hidden');
                valid = false;
            } else {
                document.getElementById('err-phone').classList.add('hidden');
            }

            if (!valid) return;

            // Ticket Object Creation
            const ticketId = 'EP-' + Math.floor(1000 + Math.random() * 9000);
            const ticketData = {
                ticketId,
                eventId: evt.id,
                eventName: evt.title,
                name,
                email,
                phone,
                tier: ticketTier,
                date: evt.displayDate,
                venue: evt.venue
            };

            registeredTickets.push(ticketData);
            localStorage.setItem('eventpulse_tickets', JSON.stringify(registeredTickets));

            updateTicketBadgeCount();
            closeEventModal();
            showDigitalPassModal(ticketData);
            showToast('Registration Successful!', 'success');
            document.getElementById('registration-form').reset();
        }

        // --- 6. DIGITAL PASS GENERATION & QR CANVAS ---
        function showDigitalPassModal(ticket) {
            document.getElementById('ticket-pass-type').innerText = ticket.tier;
            document.getElementById('ticket-event-name').innerText = ticket.eventName;
            document.getElementById('ticket-id-tag').innerText = '#' + ticket.ticketId;
            document.getElementById('ticket-attendee-name').innerText = ticket.name;
            document.getElementById('ticket-date-time').innerText = ticket.date;
            document.getElementById('ticket-venue').innerText = ticket.venue;

            drawProceduralQR(ticket.ticketId);
            document.getElementById('ticket-modal').classList.remove('hidden');
        }

        function closeTicketModal() {
            document.getElementById('ticket-modal').classList.add('hidden');
        }

        function drawProceduralQR(code) {
            const canvas = document.getElementById('qr-canvas');
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            ctx.fillStyle = '#ffffff';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            ctx.fillStyle = '#0f172a';
            const size = 5;
            for (let r = 0; r < 14; r++) {
                for (let c = 0; c < 14; c++) {
                    // Generate deterministic grid pattern based on code
                    const val = (r * 7 + c * 13 + code.charCodeAt(code.length - 1)) % 3;
                    if (val === 0 || r === 0 || c === 0 || r === 13 || c === 13) {
                        ctx.fillRect(c * size, r * size, size - 0.5, size - 0.5);
                    }
                }
            }
        }

        // --- 7. TICKET LOCKER (LOCALSTORAGE PERSISTENCE) ---
        function toggleTicketLocker() {
            const drawer = document.getElementById('ticket-locker-drawer');
            drawer.classList.toggle('hidden');
            if (!drawer.classList.contains('hidden')) {
                renderTicketLockerItems();
            }
        }

        function updateTicketBadgeCount() {
            const badge = document.getElementById('ticket-badge');
            if (registeredTickets.length > 0) {
                badge.innerText = registeredTickets.length;
                badge.classList.remove('hidden');
            } else {
                badge.classList.add('hidden');
            }
        }

        function renderTicketLockerItems() {
            const container = document.getElementById('locker-items-container');
            container.innerHTML = '';

            if (registeredTickets.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-12 text-slate-400 space-y-2">
                        <i class="fa-solid fa-ticket text-3xl"></i>
                        <p class="text-sm font-medium">No registrations saved yet.</p>
                    </div>
                `;
                return;
            }

            registeredTickets.forEach((t, index) => {
                const item = document.createElement('div');
                item.className = 'p-4 rounded-2xl bg-slate-100 dark:bg-slate-800/80 border border-slate-200 dark:border-slate-700/60 space-y-2 text-xs relative group';
                item.innerHTML = `
                    <div class="flex justify-between items-start">
                        <div>
                            <span class="px-2 py-0.5 bg-brand-600 text-white font-bold rounded text-[10px]">${t.tier}</span>
                            <h4 class="font-bold text-sm text-slate-900 dark:text-white mt-1">${t.eventName}</h4>
                        </div>
                        <button onclick="cancelRegistration(${index})" class="text-rose-500 hover:text-rose-700 p-1" title="Cancel Booking">
                            <i class="fa-solid fa-trash-can"></i>
                        </button>
                    </div>
                    <div class="text-slate-600 dark:text-slate-400 space-y-0.5">
                        <p><i class="fa-regular fa-user mr-1.5"></i>${t.name} (${t.email})</p>
                        <p><i class="fa-regular fa-calendar mr-1.5"></i>${t.date}</p>
                    </div>
                    <button onclick="viewSavedTicket(${index})" class="w-full mt-2 py-1.5 bg-slate-900 dark:bg-slate-100 text-white dark:text-slate-900 font-bold rounded-lg text-center block">
                        View QR Pass
                    </button>
                `;
                container.appendChild(item);
            });
        }

        function viewSavedTicket(index) {
            toggleTicketLocker();
            showDigitalPassModal(registeredTickets[index]);
        }

        function cancelRegistration(index) {
            if (confirm("Are you sure you want to cancel this registration?")) {
                registeredTickets.splice(index, 1);
                localStorage.setItem('eventpulse_tickets', JSON.stringify(registeredTickets));
                updateTicketBadgeCount();
                renderTicketLockerItems();
                showToast('Registration cancelled', 'info');
            }
        }

        }
